from functools import wraps

def memoize(func):
    cache = {}
    @wraps(func)
    def wrapper(*args, **kwargs):
        key = (args, tuple(sorted(kwargs.items())))
        if key not in cache:
            cache[key] = func(*args, **kwargs)
        return cache[key]
    return wrapper

@memoize
def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

assert fibonacci(0) == 0

def parse_sql(sql: str) -> dict:
    """
    解析简单 SQL: SELECT col1, col2 FROM table WHERE cond1 AND cond2
    返回: {selected_columns: [...], table_name: str, where_conditions: [...]}
    """
    sql = sql.strip().rstrip(';')

    # 匹配 SELECT ... FROM ... [WHERE ...]
    pattern = re.compile(
        r'^\s*SELECT\s+(?P<cols>.+?)\s+FROM\s+(?P<table>\w+)(?:\s+WHERE\s+(?P<where>.+?))?\s*$',
        re.IGNORECASE
    )
    m = pattern.match(sql)
    if not m:
        raise ValueError(f"无法解析 SQL: {sql}")

    # 解析列
    selected_columns = [c.strip() for c in m.group('cols').split(',')]

    # 解析表名
    table_name = m.group('table')

    # 解析 WHERE 条件
    where_conditions = []
    if m.group('where'):
        # 按 AND 分割（忽略大小写）
        conditions = re.split(r'\s+AND\s+', m.group('where'), flags=re.IGNORECASE)
        for cond in conditions:
            cond = cond.strip()
            # 匹配: column operator value
            cm = re.match(
                r'^(?P<col>\w+)\s+(?P<op>=|!=|<>|<=|>=|<|>|LIKE|IN|IS)\s+(?P<val>.+)$',
                cond, re.IGNORECASE
            )
            if cm:
                val = cm.group('val').strip()
                # 去掉引号
                if (val.startswith("'") and val.endswith("'")) or \

                })

    return {

        'table_name': table_name,
        'where_conditions': where_conditions
