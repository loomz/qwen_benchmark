from functools import wraps

def memoize(func):
    cache = {}

    @wraps(func)
    def wrapper(*args, **kwargs):
        key = (args, tuple(sorted(kwargs.items(), key=lambda item: item[0])))
        if key not in cache:
            cache[key] = func(*args, **kwargs)
        return cache[key]

    wrapper.cache = cache
    return wrapper

@memoize
def fibonacci(n):
    if n < 0:
        raise ValueError("n must be non-negative")
    if n <= 1:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

import re

def _split_conditions(conditions):
    if not conditions:
        return []
    parts = re.split(r"\s+AND\s+", conditions.strip(), flags=re.IGNORECASE)
    return [part.strip() for part in parts if part.strip()]


def parse_sql(sql):
    if not isinstance(sql, str):
        raise ValueError("SQL must be a string")

    sql = sql.strip().rstrip(";").strip()

    pattern = re.compile(
        r"^\s*SELECT\s+(?P<columns>.+?)\s+FROM\s+(?P<table>[A-Za-z_][A-Za-z0-9_]*)(?:\s+WHERE\s+(?P<conditions>.+?))?\s*$",
        re.IGNORECASE | re.DOTALL,
    )

    match = pattern.match(sql)
    if not match:
        raise ValueError(f"Cannot parse SQL: {sql}")

    columns_raw = match.group("columns")
    selected_columns = [col.strip() for col in columns_raw.split(",") if col.strip()]

    if not selected_columns:
        raise ValueError("No selected columns")

    table_name = match.group("table")
    where_conditions = _split_conditions(match.group("conditions"))

    return {
        "selected_columns": selected_columns,
        "table_name": table_name,
        "where_conditions": where_conditions,
    }
