from functools import wraps

def memoize(func):
    cache = {}

    @wraps(func)
    def wrapper(*args, **kwargs):
        key = (args, tuple(sorted(kwargs.items())))
        if key not in cache:
            cache[key] = func(*args, **kwargs)
        return cache[key]

    return wrapper

@memoize
def fibonacci(n):
    if n < 0:
        raise ValueError("n must be non-negative")
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

assert fibonacci(0) == 0

def parse_sql(sql: str) -> dict:
    if not isinstance(sql, str):
        raise TypeError("sql must be a string")

    match = _SQL_RE.match(sql)
    if not match:
        raise ValueError(f"Unsupported SQL statement: {sql!r}")

    selected_columns = [
        column.strip()
        for column in match.group("selected_columns").split(",")
        if column.strip()
    ]

    table_name = match.group("table_name").strip().strip("`\"'")

    where_raw = match.group("where_conditions")
    if where_raw:
        where_conditions = [
            condition.strip()
            for condition in re.split(r"\s+AND\s+", where_raw, flags=re.IGNORECASE)
            if condition.strip()
        ]
    else:
        where_conditions = []

    return {
        "selected_columns": selected_columns,
        "table_name": table_name,
        "where_conditions": where_conditions,
    }
