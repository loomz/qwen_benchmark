from functools import wraps

def memoize(func):
    cache = {}
    @wraps(func)
    def wrapper(*args, **kwargs):
        key = args + tuple(sorted(kwargs.items()))
        if key not in cache:
            cache[key] = func(*args, **kwargs)
        return cache[key]
    return wrapper

@memoize
def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

import re
import unittest

def parse_simple_sql(sql: str) -> dict:
    sql = sql.strip()
    pattern = r'^SELECT\s+(.*?)\s+FROM\s+(\w+)\s+WHERE\s+(.*)$'
    match = re.match(pattern, sql, re.IGNORECASE)
    if not match:
        raise ValueError("Invalid SQL format. Expected: SELECT col FROM table WHERE cond")

    cols_str, table_name, cond_str = match.groups()

    selected_columns = [c.strip() for c in cols_str.split(',')]
    where_conditions = re.split(r'\s+(?:AND|OR)\s+', cond_str.strip(), flags=re.IGNORECASE)
    where_conditions = [c.strip() for c in where_conditions if c.strip()]

    return {
        "selected_columns": selected_columns,
        "table_name": table_name,
        "where_conditions": where_conditions
    }

class TestSimpleSQLParser(unittest.TestCase):
    def test_basic(self):
        sql = "SELECT name, age FROM users WHERE age > 18"
        res = parse_simple_sql(sql)
        self.assertEqual(res["selected_columns"], ["name", "age"])
        self.assertEqual(res["table_name"], "users")
        self.assertEqual(res["where_conditions"], ["age > 18"])

    def test_multiple_conditions(self):
        sql = "SELECT * FROM products WHERE price > 100 AND stock < 50 OR category = 'electronics'"
        res = parse_simple_sql(sql)
        self.assertEqual(res["selected_columns"], ["*"])
        self.assertEqual(res["table_name"], "products")
        self.assertEqual(res["where_conditions"], ["price > 100", "stock < 50", "category = 'electronics'"])

    def test_case_insensitive(self):
        sql = "select id from orders where status = 'pending'"
        res = parse_simple_sql(sql)
        self.assertEqual(res["selected_columns"], ["id"])
        self.assertEqual(res["table_name"], "orders")
        self.assertEqual(res["where_conditions"], ["status = 'pending'"])

    def test_invalid_sql(self):
        with self.assertRaises(ValueError):
            parse_simple_sql("INSERT INTO users VALUES (1, 'test')")
